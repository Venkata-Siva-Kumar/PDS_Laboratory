
#include<stdio.h>
#define N 16
#define RADIUS 2
#define BLOCK_SIZE 8
#include<sys/time.h>
__constant__ int d_coef[RADIUS+1];
__global__ void stencil_1d(int *in,int *out)
{
	__shared__ int sm[BLOCK_SIZE+2*RADIUS];
	int gindex=threadIdx.x+blockIdx.x*blockDim.x;
	int lindex=threadIdx.x+RADIUS;
	sm[lindex]=in[gindex];
	if(threadIdx.x<RADIUS)
	{
		int tmp=gindex-RADIUS;
		if(tmp<0)
		{
			sm[lindex-RADIUS]=0;
		}
		else
		{
			sm[lindex-RADIUS]=in[gindex-RADIUS];
		}
		sm[lindex+BLOCK_SIZE]=in[gindex+BLOCK_SIZE];
	}
	__syncthreads();
	int result=0;
	int i;
	for(i=1;i<=RADIUS;i++)
	{
		result+=d_coef[i]*(sm[lindex+i]-sm[lindex-i]);
	}
	out[gindex]=result;
}

int main()
{

	cudaEvent_t start,stop;
	cudaEventCreate(&start);
	cudaEventCreate(&stop);
	cudaEventRecord(start,0);

	const int coef[]={0,1,2};
	cudaMemcpyToSymbol(d_coef,coef,(RADIUS+1)*sizeof(int));
	int i;
	int in[N+2*RADIUS],out[N];
	int *d_in,*d_out;
	cudaMalloc((void**)&d_in,(N+2*RADIUS)*sizeof(int));
	cudaMalloc((void**)&d_out,N*sizeof(int));
	for(i=0;i<(N+2*RADIUS);i++)
	{
		in[i]=i;
	}
	cudaMemcpy(d_in,in,(N+2*RADIUS)*sizeof(int),cudaMemcpyHostToDevice);
	stencil_1d<<<(N+BLOCK_SIZE-1)/BLOCK_SIZE,BLOCK_SIZE>>>(d_in,d_out);
	cudaMemcpy(out,d_out,N*sizeof(int),cudaMemcpyDeviceToHost);
	
	cudaEventRecord(stop,0);
	cudaEventSynchronize(stop);
	float elapseTime;
	cudaEventElapsedTime(&elapseTime,start,stop);
	
	printf("\n");
	for(i=0;i<(N+2*RADIUS);i++)
	{
		printf("\t%d",in[i]);
	}
	printf("\n");
	for(i=0;i<N;i++)
	{
		printf("\t%d",out[i]);
	}
	printf("\n");
	printf("Time Elapsed :%3.1f ms\n",elapseTime);
	cudaEventDestroy(start);
	cudaEventDestroy(stop);
	cudaFree(d_in);
	cudaFree(d_out);
	return 0;
}
