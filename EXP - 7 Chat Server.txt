#include<mpi.h> 
#include<stdio.h> 

int main(int argc,char *argv[]) 
{ 
	int a[2],c,e,f,d,r; 
	MPI_Init(&argc,&argv); 
	MPI_Status s; 
	MPI_Comm_rank(MPI_COMM_WORLD,&r); 
	if(r==0) 
	{ 
		printf("\n Enter two values: "); 
		scanf("%d %d",&a[0],&a[1]); 
		MPI_Bcast(&a,2,MPI_INT,0,MPI_COMM_WORLD); 
	} 
	if(r==1) 
	{ 
		MPI_Bcast(&a,2,MPI_INT,0,MPI_COMM_WORLD); 
		c=a[0]+a[1]; 
		printf("sum=%d\n",c); 
	} 
	if(r==2) 
	{ 
		MPI_Bcast(&a,2,MPI_INT,0,MPI_COMM_WORLD); 
		d=a[0]-a[1]; 
		printf("Difference=%d\n",d); 
	} 
	if(r==3) 
	{ 
		MPI_Bcast(&a,2,MPI_INT,0,MPI_COMM_WORLD); 
		e=a[0]*a[1]; 
		printf("Product=%d\n",e); 
	} 
	if(r==4) 
	{ 
		MPI_Bcast(&a,2,MPI_INT,0,MPI_COMM_WORLD); 
		f=a[0]/a[1]; 
		printf("Division=%d\n",f); 
	} 
	MPI_Finalize(); 
	return 0; 
}
