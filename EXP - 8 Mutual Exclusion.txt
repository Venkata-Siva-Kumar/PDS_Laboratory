#include<stdio.h>
#include<mpi.h>
#include<string.h>

int main()
{
	MPI_Init(NULL,NULL);
	int rank,size,indv_ts, all_ts[3],i;
	int min,minpid,count = 0;
	
	char msg[10];
	MPI_Status s;
	MPI_Comm_rank(MPI_COMM_WORLD,&rank);
	MPI_Comm_size(MPI_COMM_WORLD,&size);
	
	printf("\nEnter Timestamp for Process %d: ",rank);
	scanf("%d",&indv_ts);
	
	for(i=0;i<size;i++)
	{
		MPI_Send(&indv_ts, 1, MPI_INT, i,111,MPI_COMM_WORLD);
	}
	
	for(i=0;i<size;i++)
	{
		MPI_Recv(&all_ts[i], 1, MPI_INT, i,111,MPI_COMM_WORLD,&s);
	}
	
	min = all_ts[0];
	minpid = 0;
	for(i=1;i<size;i++)
	{
		if(all_ts[i] < min)
		{
			min = all_ts[i];
			minpid = i;
		}
	}
	strcpy(msg,"OK");
	MPI_Send(msg, sizeof(msg), MPI_CHAR, minpid, 222, MPI_COMM_WORLD);
	
	if(rank == minpid)
	{
		for(i=0;i<(size-1);i++)
		{
			MPI_Recv(msg, sizeof(msg), MPI_CHAR,MPI_ANY_SOURCE, 222, MPI_COMM_WORLD,&s);
			if(strcmp(msg,"OK")==0)
			{
				count++;
			}
		}
	
		if(count==2)
		{
			printf("\nProcess %d got permission to access shared data\n",rank);
		}
	}
	
	MPI_Finalize();
	return 0;
}
