#include<stdio.h>
#include<cuda.h>
#define N 16

__global__ void multiply(int *a,int *b,int *c)
{
	int i=threadIdx.x;
	if(i<N)
		c[i]=a[i]*b[i];
}

__global__ void add(int *d,int *e)
{
	int t=threadIdx.x;
	int psize=N/2;
	while(psize!=0)
	{
		d[t]=d[t]+d[t+psize];
		psize=psize/2;
		__syncthreads();
	}
	if(t==0)
		*e=d[0];
}

int main()
{
	cudaEvent_t start,stop;
	cudaEventCreate(&start);
	cudaEventCreate(&stop);

	int *a,*b,*c,*d,i;

	int size=N*sizeof(int);

	a=(int*)malloc(size);
	b=(int*)malloc(size);
	c=(int*)malloc(size);
	d=(int*)malloc(size);

	int *da,*db,*dc,*dd;

	cudaMalloc((void**)&da,size);
	cudaMalloc((void**)&db,size);
	cudaMalloc((void**)&dc,size);
	cudaMalloc((void**)&dd,size);
	for(i=0;i<N;i++)
	{
			a[i]=1;
			b[i]=2;
	}

	cudaMemcpy(da,a,size,cudaMemcpyHostToDevice);
	cudaMemcpy(db,b,size,cudaMemcpyHostToDevice);

	cudaEventRecord(start,0);

	multiply<<<1,N>>>(da,db,dc);
	add<<<1,N/2>>>(dc,dd);

	cudaEventRecord(stop,0);
	cudaEventSynchronize(stop);

	cudaMemcpy(d,dd,size,cudaMemcpyDeviceToHost);

	printf("\nDot Product of two vectors :%d\n",d[0]);

	float elapsedtime;
	cudaEventElapsedTime(&elapsedtime,start,stop);
	printf("CUDA Execution Time = %.9f ms",elapsedtime);
	free(a);
	free(b);
	free(c);
	free(d);

	cudaFree(da);
	cudaFree(db);
	cudaFree(dc);
	cudaFree(dd);

	cudaEventDestroy(start);
	cudaEventDestroy(stop);
	printf("\n");
	return 0;
}
